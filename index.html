document.getElementById('institution-access-form').addEventListener('submit', function(e) { e.preventDefault(); const orgName = document.getElementById('inst-org-name').value.trim(); const orgEmail = document.getElementById('inst-org-email').value.trim(); if(!orgName || !orgEmail) { showCustomModal("Missing Information", "Please enter both your organization name and email address."); return; } // Save request to pending list in localStorage const pendingRequests = JSON.parse(localStorage.getItem('imikurire_pending_requests') || '[]'); const newReq = { id: 'REQ-' + Math.floor(1000 + Math.random() * 9000), orgName: orgName, orgEmail: orgEmail, date: new Date().toLocaleString(), status: 'pending' }; pendingRequests.push(newReq); localStorage.setItem('imikurire_pending_requests', JSON.stringify(pendingRequests)); // Dispatch direct mailto notification to magnibmg@gmail.com const subject = encodeURIComponent("INSTITUTIONAL FUNDING ACCESS REQUEST: " + orgName); const body = encodeURIComponent( `Hello Magnifique,\n\nA new institution has requested access to the Imikurire Project funding credentials:\n\n` + `Organization: ${orgName}\n` + `Email: ${orgEmail}\n` + `Request ID: ${newReq.id}\n` + `Date/Time: ${newReq.date}\n\n` + `Please log in to your Magnifique Admin Portal on the website using your secret passcode (2299) to review and approve or deny this request.` ); // Open mailto link window.location.href = `mailto:magnibmg@gmail.com?subject=${subject}&body=${body}`; // Show success feedback showCustomModal("Request Logged & Email Dispatched", `Thank you, ${orgName}. Your institutional access request has been sent to Magnifique (magnibmg@gmail.com).\n\n` + `Request ID: ${newReq.id}\nStatus: Pending Admin Approval.` ); document.getElementById('institution-access-form').reset(); closeInstitutionModal(); }); // Admin Portal Logic window.openAdminModal = function() { document.getElementById('admin-login-modal').classList.remove('hidden'); }; window.closeAdminModal = function() { document.getElementById('admin-login-modal').classList.add('hidden'); document.getElementById('admin-dashboard').classList.add('hidden'); document.getElementById('admin-passcode').value = ''; }; window.verifyAdminPin = function() { const pin = document.getElementById('admin-passcode').value.trim(); if(pin === '2299') { document.getElementById('admin-login-modal').classList.add('hidden'); document.getElementById('admin-dashboard').classList.remove('hidden'); renderAdminRequests(); } else { showCustomModal("Access Denied", "Incorrect passcode. Please enter the correct PIN."); } }; window.renderAdminRequests = function() { const listContainer = document.getElementById('admin-requests-list'); const pendingRequests = JSON.parse(localStorage.getItem('imikurire_pending_requests') || '[]'); if(pendingRequests.length === 0) { listContainer.innerHTML = '

No institutional requests found at this time.

'; return; } let html = ''; pendingRequests.forEach((req, index) => { const isApproved = req.status === 'approved'; html += `
${req.orgName}
Email: ${req.orgEmail}
Requested: ${req.date} | ID: ${req.id}
${isApproved ? 'APPROVED & UNLOCKED' : 'PENDING REVIEW'}
${!isApproved ? ` ` : ` `}
`; }); listContainer.innerHTML = html; if(window.lucide) lucide.createIcons(); }; window.approveRequest = function(index) { const pendingRequests = JSON.parse(localStorage.getItem('imikurire_pending_requests') || '[]'); if(pendingRequests[index]) { pendingRequests[index].status = 'approved'; localStorage.setItem('imikurire_pending_requests', JSON.stringify(pendingRequests)); // Unlock account details for the session localStorage.setItem('imikurire_funding_unlocked', 'true'); renderAdminRequests(); showCustomModal("Access Approved", `You have approved ${pendingRequests[index].orgName}. Account details are now unlocked.`); } }; window.revokeRequest = function(index) { const pendingRequests = JSON.parse(localStorage.getItem('imikurire_pending_requests') || '[]'); if(pendingRequests[index]) { pendingRequests[index].status = 'pending'; localStorage.setItem('imikurire_pending_requests', JSON.stringify(pendingRequests)); localStorage.removeItem('imikurire_funding_unlocked'); renderAdminRequests(); } }; window.deleteRequest = function(index) { const pendingRequests = JSON.parse(localStorage.getItem('imikurire_pending_requests') || '[]'); pendingRequests.splice(index, 1); localStorage.setItem('imikurire_pending_requests', JSON.stringify(pendingRequests)); renderAdminRequests(); }; });
